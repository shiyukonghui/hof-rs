   65| // ---------------------------------------------------------------------------
   66| // The batch transaction (DECISION D1).
   67| //
   68| // A prepared element: the node exists, its name and properties are applied, but
   69| // it is **not** in the scene tree yet. The commit phase attaches every prepared
   70| // node in input order. `pending_path` is the path the node will have once
   71| // attached, computed from the parent's relative path and the (possibly
   72| // sanitised) name - a detached node has no `get_path_to` answer, and the
   73| // rollback envelope has to name what was rolled back.
   74| // ---------------------------------------------------------------------------
   75| struct _PendingNode {
   76| 	int index = 0;
   77| 	Node *parent = nullptr;
   78| 	Node *node = nullptr;
   79| 	String requested_type;
   80| 	String actual_class;
   81| 	String requested_parent_path;
   82| 	String pending_path;
   83| };

   97| // The `data.batch` envelope of a rolled-back call. Its `status` is never "ok",
   98| // its `created` list is empty and it names both the refusal and everything that
   99| // was taken back, so the wire cannot be read as a partial success.
  100| static Dictionary _rollback_envelope(const Array &p_errors, const Array &p_rolled_back) {
  101| 	Dictionary envelope;
  102| 	envelope["status"] = "rolled_back";
  103| 	envelope["created"] = Array();
  104| 	envelope["count"] = 0;
  105| 	envelope["errors"] = p_errors;
  106| 	envelope["rolled_back"] = p_rolled_back;
  107| 	envelope["on_error"] = "all_or_nothing";
  108| 	return envelope;
  109| }

  111| // The chosen wire shape (documented in REPORT-017 section 5): the JSON-RPC
  112| // `error` carries the failure code and a readable message naming the element
  113| // index, `error.data.suggestion` explains the rollback, and `error.data.batch`
  114| // is the full `{"status":"rolled_back", ...}` envelope. `result` is null because
  115| // a tool that fills `r_error` answers no payload (mcp_jsonrpc.cpp:296-303), so
  116| // a rolled-back call is impossible to read as a success.
  117| static MCPToolError _attach_batch_envelope(MCPToolError p_error, const Dictionary &p_envelope,
  118| 		const String &p_suggestion) {
  119| 	Dictionary data;
  120| 	if (p_error.data.get_type() == Variant::DICTIONARY) {
  121| 		data = p_error.data;
  122| 	}
  123| 	data["batch"] = p_envelope;
  124| 	data["suggestion"] = p_suggestion;
  125| 	p_error.data = data;
  126| 	return p_error;
  127| }

  129| // Rolls the prepare phase back (reverse order, `memdelete`, nothing is attached
  130| // yet) and builds the single refusal of the call.
  131| static Variant _transaction_fail(Vector<_PendingNode> &p_pending, MCPToolError &r_error,
  132| 		int p_index, const String &p_type, const String &p_property, const String &p_parent_path,
  133| 		const String &p_reason, int p_code, const String &p_message, const String &p_suggestion) {
  134| 	Array rolled_back;
  135| 	for (int i = p_pending.size() - 1; i >= 0; i--) {
  136| 		Dictionary entry;
  137| 		entry["index"] = p_pending[i].index;
  138| 		entry["type"] = p_pending[i].actual_class;
  139| 		entry["node_path"] = p_pending[i].pending_path;
  140| 		entry["reason"] = "transaction rollback";
  141| 		rolled_back.push_back(entry);
  142| 	}
  143| 	for (int i = p_pending.size() - 1; i >= 0; i--) {
  144| 		memdelete(p_pending[i].node);
  145| 	}
  146| 	p_pending.clear();
  147| 
  148| 	Array errors;
  149| 	errors.push_back(_batch_error_entry(p_index, p_type, p_property, p_parent_path, p_reason));
  150| 
  151| // TASK-051 C-3: is the parent this element cannot find the *requested* path of a
  152| // later element of the same batch? Only used to improve the refusal message -
  153| // the batch is refused either way - but the distinction matters to a caller:
  154| // "there is no such node" and "your parent is created later in this array" need
  155| // opposite fixes (create it first vs. reorder the array). It is also the shape a
  156| // parent/child *cycle* takes, since the batch is applied in array order and only
  157| // ever looks backwards.
  158| //
  159| // The paths compared are the elements' **requested** ones (`parent_path` +
  160| // `name`), not the prepared `pending_path`, because the elements after
  161| // `p_from_index` have not been prepared yet.
  162| static int _later_element_requesting(const Array &p_nodes, int p_from_index,
  163| 		const String &p_parent_path, const String &p_root_name, String &r_later_path) {
  164| 	for (int j = p_from_index + 1; j < p_nodes.size(); j++) {
  165| 		const Variant element = p_nodes[j];
  166| 		if (element.get_type() != Variant::DICTIONARY) {
  167| 			continue;
  168| 		}
  169| 		const Dictionary entry = element;
  170| 		const Variant name_value = entry.get("name", Variant());
  171| 		if (name_value.get_type() != Variant::STRING) {
  172| 			continue;
  173| 		}
  174| 		const String name = name_value;
  175| 		if (name.is_empty()) {
  176| 			continue;
  177| 		}
  178| 		String parent_path = ".";
  179| 		const Variant parent_value = entry.get("parent_path", Variant());
  180| 		if (parent_value.get_type() == Variant::STRING && !String(parent_value).strip_edges().is_empty()) {
  181| 			parent_path = parent_value;
  182| 		}
  183| 		const String requested = _pending_path_for(parent_path, name);
  184| 		if (_pending_path_matches(requested, p_parent_path, p_root_name)) {
  185| 			r_later_path = requested;
  186| 			return j;
  187| 		}
  188| 	}
  189| 	return -1;
  190| }

  236| // Rolls the prepare phase back (reverse order, `memdelete`, nothing is attached
  237| // yet) and builds the single refusal of the call.
  238| static Variant _transaction_fail(Vector<_PendingNode> &p_pending, MCPToolError &r_error,
  239| 		int p_index, const String &p_type, const String &p_property, const String &p_parent_path,
  240| 		const String &p_reason, int p_code, const String &p_message, const String &p_suggestion) {
  241| 	Array rolled_back;
  242| 	for (int i = p_pending.size() - 1; i >= 0; i--) {
  243| 		Dictionary entry;
  244| 		entry["index"] = p_pending[i].index;
  245| 		entry["type"] = p_pending[i].actual_class;
  246| 		entry["node_path"] = p_pending[i].pending_path;
  247| 		entry["reason"] = "transaction rollback";
  248| 		rolled_back.push_back(entry);
  249| 	}
  250| 	for (int i = p_pending.size() - 1; i >= 0; i--) {
  251| 		memdelete(p_pending[i].node);
  252| 	}
  253| 	p_pending.clear();
  254| 
  255| 	Array errors;
  256| 	errors.push_back(_batch_error_entry(p_index, p_type, p_property, p_parent_path, p_reason));
  257| 
  258| 	MCPToolError error;
  259| 	switch (p_code) {
  260| 		case MCP_ERR_NOT_FOUND:
  261| 			error = MCPToolError::not_found(p_message, p_suggestion);
  262| 			break;
  263| 		case MCP_ERR_TOOL_STATE:
  264| 			error = MCPToolError::tool_state(p_message, p_suggestion);
  265| 			break;
  266| 		default:
  267| 			error = MCPToolError::invalid_params(p_message);
  268| 			break;
  269| 	}
  270| 	r_error = _attach_batch_envelope(error, _rollback_envelope(errors, rolled_back), p_suggestion);
  271| 	return Variant();
  272| }

  436| 			MCPToolError property_error;
  437| 			const Variant written = write_node_property(node, property, properties[keys[k]], property_error);
  438| 			if (written.get_type() == Variant::NIL) {
  439| 				memdelete(node);
  440| 				return _transaction_fail(pending, r_error, i, type, property, parent_path,
  441| 						property_error.message, property_error.code,
  442| 						vformat("nodes[%d]: %s", i, property_error.message), BATCH_ROLLBACK_SUGGESTION);
  443| 			}
  444| 		}
  445| 
  446| 		// Prepared, deliberately still unattached.
  447| 		_PendingNode prepared;
  448| 		prepared.index = i;
  449| 		prepared.parent = parent;
  450| 		prepared.node = node;
  451| 		prepared.requested_type = type;
  452| 		prepared.actual_class = node->get_class();
  453| 		prepared.requested_parent_path = parent_path;
  454| 		prepared.parent_from_batch = parent_from_batch;
  455| 		const String actual_name = String(node->get_name());
  456| 		prepared.pending_path = _pending_path_for(parent_relative, actual_name);
  457| 
  458| 		// TASK-051 C-3: in the new mode a duplicate path inside one request is a
  459| 		// conflict, not an engine rename. The second element would silently
  460| 		// become a node whose name is not the one it asked for, and a later
  461| 		// element naming that path could not tell the two apart. The default mode
  462| 		// keeps the engine's own rename (the existing doctest pins it).
  463| 		if (p_resolve_within_batch && !prepared.pending_path.is_empty()) {
  464| 			for (int p = 0; p < pending.size(); p++) {
  465| 				if (pending[p].pending_path != prepared.pending_path) {
  466| 					continue;
  467| 				}
  468| 				memdelete(node);
  469| 				return _transaction_fail(pending, r_error, i, type, String(), parent_path,
  470| 						vformat("Two nodes of this batch would both be '%s' (nodes[%d] and nodes[%d])",
  471| 								prepared.pending_path, pending[p].index, i),
  472| 						MCP_ERR_INVALID_PARAMS,
  473| 						vformat("nodes[%d]: '%s' is already created by nodes[%d] of this batch, so a reference to it would be ambiguous",
  474| 								i, prepared.pending_path, pending[p].index),
  475| 						"Give every element of one parent's children a distinct 'name' (the engine would otherwise rename the second node), or split the request into two editor_add_nodes_batch calls");
  476| 			}
  477| 		}
  478| 		pending.push_back(prepared);
  479| 	}
  480| 
  481| 	// Commit phase: attach in input order, then read the real name and path back.
  482| 	Array created;
  483| 	for (int i = 0; i < pending.size(); i++) {
  484| 		const _PendingNode &prepared = pending[i];
  485| 		prepared.parent->add_child(prepared.node);
  486| 		// `owner` has to be set *after* `add_child()`: it is the ancestor
  487| 		// relationship that makes the root a legal owner.
  488| 		prepared.node->set_owner(p_root);
  489| 
  490| 		Dictionary entry;
  491| 		entry["index"] = prepared.index;
  492| 		entry["type"] = prepared.node->get_class();
  493| 		entry["name"] = String(prepared.node->get_name());
  494| 		entry["parent_path"] = prepared.requested_parent_path;
  495| 		// TASK-051 C-3: which of the two parent sources answered. It is the only
  496| 		// way to tell "attached under a node this call created" from "attached
  497| 		// under a same-named node that was already in the scene", and a client
  498| 		// feeding `node_path` back into another tool needs that distinction.
  499| 		entry["parent_source"] = prepared.parent_from_batch ? "batch" : "scene";
  500| 		entry["node_path"] = relative_path(p_root, prepared.node);
  501| 		created.push_back(entry);
  502| 	}
  503| 
  504| 	Dictionary result;
  505| 	result["status"] = "ok";
  506| 	result["created"] = created;
  507| 	result["count"] = created.size();
  508| 	result["errors"] = Array();
  509| 	// TASK-051 C-3: the mode is echoed so a caller (and a doctest) can read back
  510| 	// which parent-resolution rule the answer was produced under.
  511| 	result["resolve_within_batch"] = p_resolve_within_batch;
  512| 	return result;
  513| }

  515| Variant set_node_property_batch_on(Node *p_root, const String &p_node_type, const String &p_property,
  516| 		const Variant &p_raw_value, MCPToolError &r_error) {
  517| 	const StringName type_name(p_node_type);
  518| 	if (!ClassDB::class_exists(type_name)) {
  519| 		r_error = MCPToolError::invalid_params(vformat("Unknown node type: '%s'", p_node_type));
  520| 		return Variant();
  521| 	}
  522| 
  523| 	Vector<Node *> matched;
  524| 	collect_nodes_by_type(p_root, p_node_type, matched);
  525| 	if (matched.is_empty()) {
  526| 		// The migration source answered `{"updated": 0}` as a success here. This
  527| 		// module refuses a call that could not do what its name says.
  528| 		r_error = MCPToolError::not_found(vformat("No node of type '%s' in the edited scene", p_node_type),
  529| 				"Use editor_find_nodes_by_type to list the nodes this scene has, or editor_add_nodes_batch/add a node of that type first");
  530| 		return Variant();
  531| 	}
  532| 
  533| 	// The pre-check the migration source did not do at all: one missing property
  534| 	// on one matched node refuses the whole call *before* a single write.
  535| 	//
  536| 	// TASK-028 G-1: the question is asked about the whole **property path**
  537| 	// (`position:y` is as much a property of the node as `position` is), so the
  538| 	// batch accepts exactly what `editor_set_node_property` accepts. A path that
  539| 	// cannot be indexed at all, or that is malformed, is a request-shape
  540| 	// violation and is reported as such instead of as "the node does not declare
  541| 	// it".
  542| 	for (Node *node : matched) {
  543| 		MCPToolError path_error;
  544| 		if (node_property_path_exists(node, p_property, path_error)) {
  545| 			continue;
  546| 		}
  547| 		if (path_error.is_error() && path_error.code == -32602) {
  548| 			r_error = path_error;
  549| 			return Variant();
  550| 		}
  551| 		r_error = MCPToolError::not_found(
  552| 				vformat("Property '%s' on node '%s'", p_property, relative_path(p_root, node)),
  553| 				"Every matched node must declare the property. Use editor_get_node_properties on the nodes "
  554| 				"editor_find_nodes_by_type returns to see which one does not");
  555| 		return Variant();
  556| 	}
  557| 
  558| 	// TASK-018 section 1.2: the *value* is validated against every matched node
  559| 	// as well, before the first write. A layout-incompatible value
  560| 	// (`{"node_type":"Node2D","property":"position","value":1e20}`) is a
  561| 	// validation failure, so it must be caught here rather than by the first
  562| 	// write inside the loop below: `prepare_node_property_value` is the exact
  563| 	// validation half of `write_node_property`, so "this value fits" has one
  564| 	// definition and the whole call is refused with nothing written on any node.
  565| 	for (Node *node : matched) {
  566| 		Variant converted;
  567| 		MCPToolError value_error;
  568| 		if (!prepare_node_property_value(node, p_property, p_raw_value, converted, value_error)) {
  569| 			value_error.message = vformat(
  570| 					"Property '%s' write refused before any node was written: %s", p_property, value_error.message);
  571| 			r_error = value_error;
  572| 			return Variant();
  573| 		}
  574| 	}
  575| 
  576| 	Vector<Node *> written;
  577| 	Vector<Variant> previous_values;
  578| 	// TASK-037 D2 (self-audit): the same read-back question the single-node
  579| 	// writer answers, asked per matched node. `write_node_property` returns the
  580| 	// engine's `ignored` bag for the node it just wrote; a batch that answered
  581| 	// only `{"updated": N, "status": "ok"}` hid the fact that the engine's own
  582| 	// setter clamped the value on some (or every) node - the exact shape
  583| 	// `editor_set_node_property(path=Sprite, property=hframes, value=0)` exposes
  584| 	// live (`new_value: 1`). The bag is keyed by the node's relative path so one
  585| 	// entry is still readable, and `ignored_count` says at a glance whether the
  586| 	// call stored the request everywhere.
  587| 	Dictionary ignored;
  588| 	for (Node *node : matched) {
  589| 		// TASK-028 G-1: the value to put back is read through the same property
  590| 		// path the write uses, so a sub-property write is rolled back to the
  591| 		// component's own previous value instead of to nothing.
  592| 		bool previous_valid = false;
  593| 		MCPToolError previous_error;
  594| 		const Variant previous = read_node_property_path(node, p_property, previous_valid, previous_error);
  595| 		MCPToolError write_error;
  596| 		const Variant written_result = write_node_property(node, p_property, p_raw_value, write_error);
  597| 		if (written_result.get_type() == Variant::NIL) {
  598| 			// Restore what this call already changed, newest first, so a failure
  599| 			// in the middle leaves no node written.
  600| 			for (int j = written.size() - 1; j >= 0; j--) {
  601| 				MCPToolError rollback_error;
  602| 				apply_node_property_value(written[j], p_property, previous_values[j], rollback_error);
  603| 			}
  604| 			write_error.message = vformat(
  605| 					"Property '%s' write failed after %d node(s) had been written; their previous values were restored: %s",
  606| 					p_property, written.size(), write_error.message);
  607| 			r_error = write_error;
  608| 			return Variant();
  609| 		}
  610| 		const Dictionary node_answer = written_result;
  611| 		const Dictionary node_ignored = node_answer.has("ignored") ? (Dictionary)node_answer["ignored"] : Dictionary();
  612| 		if (node_ignored.has(p_property)) {
  613| 			ignored[relative_path(p_root, node)] = node_ignored[p_property];
  614| 		}
  615| 		written.push_back(node);
  616| 		previous_values.push_back(previous);
  617| 	}
  618| 
  619| 	Array paths;
  620| 	for (Node *node : written) {
  621| 		paths.push_back(relative_path(p_root, node));
  622| 	}
  623| 
  624| 	Dictionary result;
  625| 	result["updated"] = written.size();
  626| 	result["property"] = p_property;
  627| 	result["node_type"] = p_node_type;
  628| 	result["status"] = "ok";
  629| 	result["nodes"] = paths;
  630| 	result["count"] = written.size();
  631| 	// Per-node `{requested, stored, reason}` for every node whose write the engine
  632| 	// stored differently (TASK-037 D2; see the comment above). Empty means the
  633| 	// engine stored the request on every matched node.
  634| 	result["ignored"] = ignored;
  635| 	result["ignored_count"] = ignored.size();
  636| 	return result;
  637| }

  641| // ---------------------------------------------------------------------------
  642| // editor_add_nodes_batch (old `batch_add_nodes`, batch.rs:270)
  643| //
  644| // The argument grammar is validated here (array-ness, non-emptiness) so a
  645| // malformed call is `-32602` even in a process without an edited scene; the
  646| // transaction itself is `MCPTools::add_nodes_batch_on`.
  647| //
  648| // Wire shape of a rolled-back call (the decision recorded in REPORT-017):
  649| //   * `error.code`  - `-32602` for a shape/type violation, `-32001` for a
  650| //     missing parent or property;
  651| //   * `error.message` - a readable message naming the element index;
  652| //   * `error.data.suggestion` - the rollback explanation;
  653| //   * `error.data.batch` - `{"status":"rolled_back","created":[],"count":0,
  654| //     "errors":[{index,type,property,parent_path,reason}],
  655| //     "rolled_back":[{index,type,node_path,reason}],"on_error":"all_or_nothing"}`;
  656| //   * `result` is null (the handler answers no payload).
  657| // ---------------------------------------------------------------------------
  658| static Variant _tool_add_nodes_batch(const Dictionary &p_args, MCPToolError &r_error) {
  659| 	const Variant nodes_value = p_args.get("nodes", Variant());
  660| 	if (nodes_value.get_type() == Variant::NIL) {
  661| 		r_error = MCPToolError::invalid_params("Missing required parameter: nodes");
  662| 		return Variant();
  663| 	}
  664| 	if (nodes_value.get_type() != Variant::ARRAY) {
  665| 		r_error = MCPToolError::invalid_params(vformat("'nodes' must be an array, got %s",
  666| 				Variant::get_type_name(nodes_value.get_type())));
  667| 		return Variant();
  668| 	}
  669| 	const Array nodes = nodes_value;
  670| 	if (nodes.is_empty()) {
  671| 		r_error = MCPToolError::invalid_params("Parameter 'nodes' must be a non-empty array");
  672| 		return Variant();
  673| 	}
  674| 	// TASK-051 C-3: the mode is read (and a present-but-wrong type refused)
  675| 	// *before* the editor guard, exactly like the array shape above, so a
  676| 	// mistyped `resolve_within_batch` is a `-32602` in every process rather than
  677| 	// a `-32000` that hides it (PLAYBOOK section 6.2).
  678| 	bool resolve_within_batch = false;
  679| 	if (!optional_bool(p_args, "resolve_within_batch", false, resolve_within_batch, r_error)) {
  680| 		return Variant();
  681| 	}
  682| 	if (!require_editor_ui(r_error, "editor node writes outside a running editor",
  683| 				"Start the MCP server inside the Godot editor to write editor state")) {
  684| 		return Variant();
  685| 	}
  686| 	Node *root = edited_scene_root();
  687| 	if (root == nullptr) {
  688| 		r_error = MCPToolError::no_scene();
  689| 		return Variant();
  690| 	}
  691| 	return add_nodes_batch_on(root, nodes, r_error, resolve_within_batch);
  692| }
