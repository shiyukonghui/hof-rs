   68| 	const Variant value = p_args.get("properties", Variant());

   99| // TASK-026 (E-9, third surface of E-3): the three steps every *other* write path
  100| // of the module takes.
  101| //
  102| // `property_value_from_json` -> `shape_vector_from_json` -> `coerce_to_property_type`
  103| // is the exact sequence of `MCPTools::prepare_node_property_value`
  104| // (tools/running_game_node_write.cpp:664-680), which the editor node writer, the
  105| // batch writer, the cross-scene writer and `project_set_setting` all use. The
  106| // two resource writers used only the first and the third step, so every value
  107| // the module's own read side answers as an *object* or as an array of objects -

  111| // was accepted on the node path. That is section 23.4's closure rule failing

  113| // shapes (TASK-026 E-9) and `project_edit_resource` has to take them back.

  115| // `shape_vector_from_json` is a no-op for a target that is not component-shaped,
  116| // so nothing else about these two tools changes.

  119| // `Object.script`, ...) is taken by the same `coerce_to_property_type` OBJECT
  120| // branch the node path uses, and the class the property declares is passed along

  156| // ---------------------------------------------------------------------------
  157| // TASK-022 D-5: "was it really set?" is a **read-back** question.
  158| //
  159| // `project_create_resource` used to answer `properties_set: ["min_value"]` for
  160| // `Curve.min_value = 5.0` and report nothing else, while the engine's own setter
  161| // had stored `max_value - 0.01` (0.99) - the M4b re-audit measured it and noted
  162| // that the sibling `project_edit_resource` answers the same call with a real
  163| // `changed: {min_value: {old, new}}`. The two entries now have one standard: the
  164| // value is read back after `Object::set()` and compared with what the caller
  165| // asked for, and an attribute the setter did not store as asked is reported as
  166| // `ignored` instead of as "set".
  167| //
  168| // The comparison is made at the width the member really stores. A single
  169| // precision member keeps the nearest `float`, so `(double)(float)0.1` is not
  170| // `0.1`; comparing doubles would report a perfectly good write as "ignored". For
  171| // every other target the serialized images are compared directly (a vector is a
  172| // component object on both sides, so the comparison is a structural one).
  173| //
  174| // MCP-NARROWING: G24-RESOURCE-SET-WIDTH (TASK-023 D-7). The `(real_t)` casts
  175| // below narrow a serialized `double` to the member's width; they are safe
  176| // **because they are the comparison's own width**, and they cannot lose a value
  177| // the write gate has already refused: `properties` is coerced through
  178| // `coerce_to_property_type` (the `REAL_T` slot) *before* `Object::set()`, so the
  179| // requested side is a value the 32-bit slot can hold, and both sides are then
  180| // narrowed to that one width. A caller that could reach this line with `1e300`
  181| // would get the honest answer "the stored value is not the request" - which is
  182| // what the `ignored` entry of the D-5 read-back exists to say.
  183| // ---------------------------------------------------------------------------
  184| static bool _stored_value_matches_request(const Variant &p_requested_serialized,
  185| 		const Variant &p_stored_serialized, Variant::Type p_target_type) {
  186| 	if (p_target_type == Variant::FLOAT) {
  187| 		// MCP-NARROWING: G24-RESOURCE-SET-WIDTH - the two `(real_t)` casts below
  188| 		// are the comparison's own width, and the requested side reached this
  189| 		// function through the `REAL_T` gate (TASK-023 D-7).
  190| 		return (real_t)p_requested_serialized == (real_t)p_stored_serialized;
  191| 	}
  192| 	return p_requested_serialized == p_stored_serialized;
  193| }

  202| // ---------------------------------------------------------------------------
  203| // Atomic publish (TASK-007 section 3.1)
  204| //
  205| // `ResourceSaver::save()` writes straight into the destination, so a failure in
  206| // the middle of a save (a full disk, a format saver that refuses the value, a
  207| // write error) can leave a truncated - that is, **corrupted** - file where the
  208| // caller's resource used to be. The migration source did exactly that.
  209| //
  210| // TASK-008 section 3 hoisted the mechanism into `tools/tool_helpers.*`
  211| // (`publish_file_atomically`, `temporary_sibling_path`) so that the editor write
  212| // group's `editor_save_scene` shares it instead of growing a second, subtly
  213| // different copy. What is left here is the same private name the four tools of
  214| // this group already call, plus the one-line adapter that lets the shared helper
  215| // drive `ResourceSaver::save()`:
  216| //
  217| //   * the temporary file name keeps its extension last (`created.mcp-tmp.tres`),
  218| //     because `ResourceFormatSaver::recognize_path()` compares the text after the
  219| //     **last** dot against the saver's extension list (a measured bug of an
  220| //     earlier revision, REPORT-007 section 8.1);
  221| //   * the destination is only touched after a *successful* save of a temporary
  222| //     sibling that really exists, the old bytes are copied aside first, and a
  223| //     failed publish puts them back.
  224| //
  225| // On every failure path the temporary file (and the backup) is removed: the
  226| // group may not leave a half-written artefact behind.
  227| // ---------------------------------------------------------------------------
  228| 
  229| static Error _resource_save_writer(const String &p_temp_path, void *p_userdata) {
  230| 	const Ref<Resource> *resource = static_cast<const Ref<Resource> *>(p_userdata);
  231| 	return ResourceSaver::save(*resource, p_temp_path);
  232| }

  234| static Error _save_resource_atomically(const Ref<Resource> &p_resource, const String &p_path) {
  235| 	// The `Ref` is copied into a local so that the callback has a stable address
  236| 	// for the whole call; `publish_file_atomically` never stores it.
  237| 	Ref<Resource> held = p_resource;
  238| 	return publish_file_atomically(p_path, _resource_save_writer, &held);
  239| }
