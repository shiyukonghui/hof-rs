   73| // The editor guard
   74| //
   75| // This is the same guard as `_require_editor_ui` of
   76| // `tools/editor_read_scene_inspector.cpp`, repeated here on purpose: a group may
   77| // not edit another group's file (PLAYBOOK section 2.4 / 17.1), and the shared
   78| // `tool_helpers.*` hoist of TASK-008 section 3 is scoped to the atomic publish
   79| // helper. The two copies are byte-identical logic; REPORT-008 lists the second
   80| // copy as a candidate for a later hoist.
   81| //
   82| // `EditorInterface` is created by `register_editor_types()` and therefore exists
   83| // in *every* tools process, including `--test` and a headless tools build that
   84| // never started an editor. Its accessors, however, dereference
   85| // `EditorNode::get_singleton()` (editor_interface.cpp:105-107) without a null
   86| // check, so asking it for the selection in such a process is a null dereference.
   87| // The guard is therefore the singleton itself, not only `is_editor_process()`.
   88| // ---------------------------------------------------------------------------
   89| static bool _require_editor_ui(MCPToolError &r_error) {
   90| #ifdef MCP_EDITOR_TOOLS_ENABLED
   91| 	if (is_editor_process()) {
   92| 		if (EditorNode::get_singleton() != nullptr) {
   93| 			return true;
   94| 		}
   95| 		r_error = MCPToolError::not_implemented("the editor UI (no EditorNode is running in this process)",
   96| 				"Start the MCP server inside the Godot editor to write editor state");
   97| 		return false;
   98| 	}
   99| #endif
  100| 	r_error = MCPToolError::not_implemented("editor writes outside a running editor",
  101| 			"Run the MCP server inside the Godot editor: the editor_* tools are editor-only");
  102| 	return false;
  103| }

  113| static Node *_edited_scene_root() {
  114| 	SceneTree *tree = SceneTree::get_singleton();

  116| 		return nullptr;
  117| 	}
  118| 	return tree->get_edited_scene_root();

  121| // The migration source's `find_node` (node.rs:148-165): "." and the bare root
  122| // name mean the root, an existing relative path is resolved, and a path prefixed
  123| // with the root's name is retried without it. Unlike `Node::get_node`, a miss is
  124| // `nullptr` instead of an engine error.
  125| static Node *_find_node(Node *p_root, const String &p_path) {
  126| 	if (p_path == "." || p_path == String(p_root->get_name())) {
  127| 		return p_root;
  128| 	}
  129| 	const NodePath requested(p_path);
  130| 	if (p_root->has_node(requested)) {
  131| 		return p_root->get_node(requested);
  132| 	}
  133| 	const String root_name = String(p_root->get_name());
  134| 	const String prefix = root_name + "/";
  135| 	if (p_path.begins_with(prefix)) {
  136| 		const NodePath nested(p_path.substr(prefix.length()));
  137| 		if (p_root->has_node(nested)) {
  138| 			return p_root->get_node(nested);
  139| 		}
  140| 	}
  141| 	return nullptr;
  142| }

  144| // The selection entry of one node, with the migration source's serialisation
  145| // rules (node.rs:601-622): the root is spelled ".", a node that is neither the
  146| // root nor a descendant of it is skipped.
  147| static bool _selection_entry(Node *p_root, Node *p_node, Dictionary &r_out) {
  148| 	if (p_node == nullptr) {
  149| 		return false;
  150| 	}
  151| 	if (p_node != p_root && !p_root->is_ancestor_of(p_node)) {
  152| 		return false;
  153| 	}
  154| 	r_out["name"] = String(p_node->get_name());
  155| 	r_out["path"] = p_node == p_root ? String(".") : String(p_root->get_path_to(p_node));
  156| 	r_out["type"] = p_node->get_class();
  157| 	return true;
  158| }

  160| static Array _selection_entries(Node *p_root, const List<Node *> &p_nodes) {
  161| 	Array out;
  162| 	for (Node *node : p_nodes) {
  163| 		Dictionary entry;
  164| 		if (_selection_entry(p_root, node, entry)) {
  165| 			out.push_back(entry);
  166| 		}
  167| 	}
  168| 	return out;
  169| }

  171| // `fov` is the one optional *number* of this group; there is no `optional_float`
  172| // in the shared argument helpers. A present-but-not-a-number value is -32602
  173| // (PLAYBOOK section 6.2), never a silently ignored argument.
  174| static bool _optional_number(const Dictionary &p_args, const String &p_key, bool &r_present, double &r_out, MCPToolError &r_error) {
  175| 	const Variant value = p_args.get(p_key, Variant());
  176| 	r_present = value.get_type() != Variant::NIL;
  177| 	if (!r_present) {
  178| 		return true;
  179| 	}
  180| 	if (value.get_type() != Variant::FLOAT && value.get_type() != Variant::INT) {
  181| 		r_error = MCPToolError::invalid_params(vformat("Parameter '%s' must be a number, got %s",
  182| 				p_key, Variant::get_type_name(value.get_type())));
  183| 		return false;
  184| 	}
  185| 	r_out = (double)value;
  186| 	return true;
  187| }

  189| // An optional `{x, y, z}` object. The migration source reads each component with
  190| // `as_f64().unwrap_or(0.0)` (a silently ignored mistake); a component that is
  191| // present but not a number is refused here instead (PLAYBOOK section 6.2), while
  192| // an absent component keeps the reference's 0.0.
  193| static bool _optional_vector3(const Dictionary &p_args, const String &p_key, bool &r_present, Vector3 &r_out, MCPToolError &r_error) {
  194| 	const Variant value = p_args.get(p_key, Variant());
  195| 	r_present = value.get_type() != Variant::NIL;
  196| 	if (!r_present) {
  197| 		return true;
  198| 	}
  199| 	if (value.get_type() != Variant::DICTIONARY) {
  200| 		r_error = MCPToolError::invalid_params(vformat("Parameter '%s' must be an object with x, y, z, got %s",
  201| 				p_key, Variant::get_type_name(value.get_type())));
  202| 		return false;
  203| 	}
  204| 	const Dictionary source = value;
  205| 	const char *const axes[3] = { "x", "y", "z" };
  206| 	double components[3] = { 0.0, 0.0, 0.0 };
  207| 	for (int i = 0; i < 3; i++) {
  208| 		const Variant component = source.get(axes[i], Variant());
  209| 		if (component.get_type() == Variant::NIL) {
  210| 			continue;
  211| 		}
  212| 		if (component.get_type() != Variant::FLOAT && component.get_type() != Variant::INT) {
  213| 			r_error = MCPToolError::invalid_params(vformat("Parameter '%s.%s' must be a number, got %s",
  214| 					p_key, axes[i], Variant::get_type_name(component.get_type())));
  215| 			return false;
  216| 		}
  217| 		components[i] = (double)component;
  218| 	}
  219| 	r_out = Vector3((real_t)components[0], (real_t)components[1], (real_t)components[2]);
  220| 	return true;
  221| }

  223| // The JSON value of a resource property, converted the way the migration source
  224| // converts it (`parse_value_for_property`, serialize.rs:268-320).
  225| //
  226| // `tools/project_write_resource_scene.cpp` owns the full version of this
  227| // conversion, including the "fold an integral JSON number back to INT when the
  228| // property is an int" step and the final `@GlobalScope.type_convert`. A group may
  229| // not call another group's file-private helper (PLAYBOOK section 2.4), so the
  230| // subset this one tool needs is repeated here; the string grammar
  231| // (`#rrggbb`, `Vector2(..)` / `Vector3(..)`) and the "an object stays a
  232| // Dictionary" rule are the same.
  233| static Variant _property_value_from_json(const Variant &p_value, Variant::Type p_target_type) {
  234| 	switch (p_value.get_type()) {
  235| 		case Variant::NIL: {
  236| 			return Variant();
  237| 		}
  238| 		case Variant::BOOL: {
  239| 			return (bool)p_value;
  240| 		}
  241| 		case Variant::INT: {
  242| 			// Godot's JSON parser has a single number type, so an integral value
  243| 			// arrives as a float and is folded back when the property is an int.
  244| 			return p_target_type == Variant::INT ? Variant((int64_t)p_value) : Variant((double)p_value);
  245| 		}
  246| 		case Variant::FLOAT: {
  247| 			const double number = (double)p_value;
  248| 			if (p_target_type == Variant::INT && Math::is_finite(number) && (double)(int64_t)number == number) {
  249| 				return Variant((int64_t)number);
  250| 			}
  251| 			return Variant(number);
  252| 		}
  253| 		case Variant::STRING: {
  254| 			const String text = ((String)p_value).strip_edges();
  255| 			if (text.begins_with("#")) {
  256| 				const Color color = Color::from_string(text, Color());
  257| 				if (color != Color()) {
  258| 					return Variant(color);
  259| 				}
  260| 			}
  261| 			if (text.begins_with("Vector2(") || text.begins_with("Vector2i(") ||
  262| 					text.begins_with("Vector3(") || text.begins_with("Vector3i(") ||
  263| 					text.begins_with("Vector4(")) {
  264| 				Variant from_string;
  265| 				Variant::construct_from_string(text, from_string);
  266| 				if (from_string.get_type() != Variant::NIL) {
  267| 					return from_string;
  268| 				}
  269| 			}
  270| 			return (String)p_value;
  271| 		}
  272| 		case Variant::DICTIONARY: {
  273| 			const Dictionary source = p_value;
  274| 			Dictionary out;
  275| 			const Array keys = source.keys();
  276| 			for (int i = 0; i < keys.size(); i++) {
  277| 				out[keys[i]] = _property_value_from_json(source[keys[i]], Variant::NIL);
  278| 			}
  279| 			return out;
  280| 		}
  281| 		case Variant::ARRAY: {
  282| 			const Array source = p_value;
  283| 			Array out;
  284| 			for (int i = 0; i < source.size(); i++) {
  285| 				out.push_back(_property_value_from_json(source[i], Variant::NIL));
  286| 			}
  287| 			return out;
  288| 		}
  289| 		default: {
  290| 			return p_value;
  291| 		}
  292| 	}
  293| }

  295| // The declared type of one of the object's properties, or `Variant::NIL` when the
  296| // object does not have it.
  297| static Variant::Type _property_type(const Object *p_object, const StringName &p_name) {
  298| 	List<PropertyInfo> properties;
  299| 	p_object->get_property_list(&properties);
  300| 	for (const PropertyInfo &property : properties) {
  301| 		if (property.name == p_name) {
  302| 			return property.type;
  303| 		}
  304| 	}
  305| 	return Variant::NIL;
  306| }
  307| 
  308| static bool _coerce_to_property_type(const Variant &p_value, Variant::Type p_target_type, Variant &r_out, MCPToolError &r_error) {
  309| 	if (p_value.get_type() == Variant::FLOAT) {
  310| 		const double number = (double)p_value;
  311| 		if (Math::is_nan(number) || Math::is_inf(number)) {
  312| 			r_error = MCPToolError::invalid_params("Parameter 'resource_properties' contains a non-finite number");
  313| 			return false;
  314| 		}
  315| 	}
  316| 	if (p_target_type == Variant::NIL || p_value.get_type() == p_target_type) {
  317| 		r_out = p_value;
  318| 		return true;
  319| 	}
  320| 	r_out = VariantUtilityFunctions::type_convert(p_value, p_target_type);
  321| 	return true;
  322| }

  325| // editor_open_scene (old `open_scene`, scene.rs:117)
  326| //
  327| // `path` is required and must address the project. The reference only checked
  328| // `FileAccess::file_exists` and then called `open_scene_from_path`, reporting
  329| // success unconditionally - so a scene file that exists but cannot be *loaded*
  330| // was answered with `{"opened": true}`. That is the fake-success shape TASK-008
  331| // section 2 forbids, and `editor_open_scene` is the tool the state-chain evidence
  332| // leans on, so the result is verified: after the call the edited scene must
  333| // really be the requested file, otherwise -32603.
  334| // ---------------------------------------------------------------------------
  335| static Variant _tool_open_scene(const Dictionary &p_args, MCPToolError &r_error) {
  336| 	String raw_path;
  337| 	if (!require_string(p_args, "path", raw_path, r_error)) {
  338| 		return Variant();
  339| 	}
  340| 	String path;
  341| 	if (!normalize_project_path(raw_path, path, r_error)) {
  342| 		return Variant();
  343| 	}
  344| 	if (!_require_editor_ui(r_error)) {
  345| 		return Variant();
  346| 	}
  347| 	if (!FileAccess::exists(path)) {
  348| // `set_addon_plugin_enabled("godot_mcp", ...)` - a no-op that prints an engine
  349| // error - and then answer `{"reloading": true}`. That is exactly the fake success
  350| // TASK-008 section 2(c) forbids, so the mechanism is kept and widened to the
  351| // plugins that actually exist: every addon in `editor_plugins/enabled` is
  352| // disabled and enabled again. When that list is empty there is genuinely nothing
  353| // to reload and the tool says so with -32000 + `data.suggestion` instead of
  354| // claiming a reload it did not perform.
  355| // ---------------------------------------------------------------------------
  356| static Variant _tool_reload_plugin(const Dictionary &p_args, MCPToolError &r_error) {
  357| 	(void)p_args;
  358| 	if (!require_editor_ui(r_error, "editor writes outside a running editor",
  359| 				"Start the MCP server inside the Godot editor to write editor state")) {
  360| 		return Variant();
  361| 	}
  362| #ifdef MCP_EDITOR_TOOLS_ENABLED
  363| 	Array names;
  364| 	ProjectSettings *settings = ProjectSettings::get_singleton();
  365| 	if (settings != nullptr && settings->has_setting("editor_plugins/enabled")) {
  366| 		const Variant enabled = settings->get("editor_plugins/enabled");
  367| 		if (enabled.get_type() == Variant::PACKED_STRING_ARRAY) {
  368| 			const PackedStringArray list = enabled;
  369| 			for (int i = 0; i < list.size(); i++) {
  370| 				names.push_back(list[i]);
  371| 			}
  372| 		} else if (enabled.get_type() == Variant::ARRAY) {
  373| 			const Array list = enabled;
  374| 			for (int i = 0; i < list.size(); i++) {
  375| 				names.push_back((String)list[i]);
  376| 			}
  377| 		}
  378| 	}
  379| 
  380| 	if (names.is_empty()) {
  381| 		r_error = MCPToolError::tool_state("No editor addon plugin is enabled in this project, so there is no plugin to reload",
  382| 				"The MCP server is a built-in module and cannot be unloaded at runtime; enable an addon under Project > Project Settings > Plugins and call editor_reload_plugin again");
  383| 		return Variant();
  384| 	}
  385| 
  386| 	EditorNode *editor_node = EditorNode::get_singleton();
  387| 	if (editor_node == nullptr) {
  388| 		r_error = MCPToolError::not_implemented("the editor UI (no EditorNode is running in this process)",
  389| 				"Start the MCP server inside the Godot editor to reload plugins");
  390| 		return Variant();
  391| 	}
  392| 	for (int i = 0; i < names.size(); i++) {
  393| 		const String name = names[i];
  394| 		// `p_config_changed = false`: reloading is not a settings edit, so the
  395| 		// project file is not rewritten twice per plugin.
  396| 		editor_node->set_addon_plugin_enabled(name, false, false);
  397| 		editor_node->set_addon_plugin_enabled(name, true, false);
  398| 	}
  399| 
  400| 	Dictionary result;
  401| 	result["reloading"] = true;
  402| 	result["plugins"] = names;
  403| 	result["message"] = String::utf8("编辑器插件已禁用并重新启用");
  404| 	return result;
  405| #endif
  406| 	return Variant();
  407| }

 1000| // The authoritative `description` and `inputSchema` of each tool are the contract
 1001| // entries of docs/tools_list.renamed.json, character for character. The schemas
 1002| // are *parsed* from the exact contract JSON instead of being rebuilt as a
 1003| // hand-written Dictionary, because the gate compares all three fields verbatim.
 1004| //
 1005| // Unlike `editor_read_scene_inspector.cpp` this group's schemas contain no
 1006| // numbers at all (no `default`, no `minimum`), so the integral-number folding
 1007| // that file needs to survive Godot's single number type has nothing to fold here
 1008| // and is not repeated.
 1009| static Dictionary _schema_from_json(const char *p_json) {
 1010| 	JSON json;
 1011| 	if (json.parse(String::utf8(p_json)) != OK) {
 1012| 		ERR_PRINT("MCPTools: invalid inputSchema literal in editor_write_scene_editor.cpp");
 1013| 		return Dictionary();
 1014| 	}
 1015| 	return json.get_data();
 1016| }
