   55| static bool _try_parse_port(const String &p_text, int &r_port) {
   56| 	if (!p_text.is_valid_int()) {
   57| 		return false;
   58| 	}
   59| 	const int parsed = p_text.to_int();
   60| static const int DEFAULT_EDITOR_PORT = 9877;
   61| static const int MAX_PORT = 65535;
   62| 
   63| static bool _try_parse_port(const String &p_text, int &r_port) {
   64| 	if (!p_text.is_valid_int()) {
   65| 		return false;
   66| 	}
   67| 	const int parsed = p_text.to_int();
   68| 	if (parsed < 0 || parsed > MAX_PORT) {
   69| 		return false;
   70| 	}
   71| 	r_port = parsed;
   72| 	return true;
   73| }
   74| 
   75| MCPPortConfig parse(const Vector<String> &p_cmdline_args, bool p_has_setting, int p_setting_port, bool p_is_editor) {
   76| 	MCPPortConfig config;
   77| 	// Default: the editor listens on 9877 (same port as the GDExtension addon),
   78| 	// while a game process does not listen at all unless asked to.
   79| 	config.port = p_is_editor ? DEFAULT_EDITOR_PORT : 0;
   80| 
   81| 	int explicit_port = -1;
   82| 	const int arg_count = p_cmdline_args.size();
   83| 	for (int i = 0; i < arg_count; i++) {
   84| 		const String &arg = p_cmdline_args[i];
   85| 		if (arg.begins_with("--mcp-port=")) {
   86| 			int parsed = 0;
   87| 			if (_try_parse_port(arg.substr(String("--mcp-port=").length()), parsed)) {
   88| 				explicit_port = parsed;
   89| 			}
   90| 		} else if (arg == "--mcp-port") {
   91| 			// `--mcp-port N` form.
   92| 			if (i + 1 < arg_count) {
   93| 				int parsed = 0;
   94| 				if (_try_parse_port(p_cmdline_args[i + 1], parsed)) {
   95| 					explicit_port = parsed;
   96| 				}
   97| 				i++;
   98| 			}
   99| 		}
  100| 	}
  101| 
  102| 	if (p_has_setting && p_setting_port > 0 && p_setting_port <= MAX_PORT) {
  103| 		config.port = p_setting_port;
  104| 		config.from_project_setting = true;
  105| 	}
  106| 
  107| 	if (explicit_port >= 0) {
  108| 		config.port = explicit_port;
  109| 		config.explicit_cmdline = true;
  110| 	}
  111| 
  112| 	return config;
  113| }
  114| 
  115| bool should_listen(bool p_is_editor, const MCPPortConfig &p_config, bool p_enabled_in_game) {
  116| 	if (p_config.port <= 0) {
  117| 		return false;
  118| 	}
  119| 	if (p_is_editor) {
  120| 		return true;
  121| 	}
  122| 	// A game process must opt in explicitly (REQUIREMENTS C4).
  123| 	return p_config.explicit_cmdline || p_enabled_in_game;
  124| }

  130| int effective_ms(int p_configured_ms) {
  131| 	// TASK-014 R-3: `<= 0` is not a deadline. The transport reads 0 as "no
  132| 	// deadline" and the frame-waiting deferred tools declare no deadline of their
  133| 	// own, so a configured 0 would leave them able to hang forever - the exact
  134| 	// failure GDR-20 point 4 removes. A positive value is passed through
  135| 	// unchanged, so the setting stays a working knob.
  136| 	return p_configured_ms <= 0 ? DEFAULT_MS : p_configured_ms;
  137| }

  180| void MCPServer::_retire_unattached() {
  181| 	// Disable the module for good and release the instance. A Node that can
  182| 	// never enter a tree must not stay alive: `Node::print_orphan_nodes()`
  183| 	// reports every tree-less Node in ObjectDB and the engine's own unit-test
  184| 	// suite asserts on the exact output of one such probe, so leaving this

  186| 	MCPServer *server = singleton;
  187| 	singleton = nullptr;

  189| void MCPServer::bootstrap() {
  190| 	if (service_started) {
  191| 		return;
  192| 	}
  193| 
  194| 	SceneTree *tree = SceneTree::get_singleton();
  195| 	if (tree == nullptr || tree->get_root() == nullptr) {
  196| 		// SCENE level initialization happens before the SceneTree exists, so
  197| 		// the very first frames may still be too early.
  198| 		bootstrap_attempts++;
  199| 		if (bootstrap_attempts > 600) {
  200| 			ERR_PRINT("[MCP] SceneTree never became available; MCP server disabled.");
  201| 			// This is the engine's `--test` harness, which has no SceneTree at
  202| 			// all: retire the instance instead of leaving a tree-less Node
  203| 			// behind (see _retire_unattached).
  204| 			if (MessageQueue::get_singleton() != nullptr) {
  205| 				MessageQueue::get_singleton()->push_callable(callable_mp_static(&MCPServer::_retire_unattached));
  206| 			} else {
  207| 				_retire_unattached();
  208| 			}
  209| 			return;
  210| 		}
  211| 		if (MessageQueue::get_singleton() != nullptr) {
  212| 			MessageQueue::get_singleton()->push_callable(callable_mp(this, &MCPServer::bootstrap));
  213| 		}
  214| 		return;
  215| 	}
  216| 
  217| 	set_name("MCPServer");
  218| 	tree->get_root()->add_child(this);
  219| 	set_process(true);
  220| 
  221| 	_start_service();
  222| }

  224| void MCPServer::pump_frame(double p_delta) {
  225| 	(void)p_delta;
  226| 	// GDR-20 point 7: the frame clock of the deferred channel is the SceneTree's
  227| 	// frame counter, not a private counter of this node - the tools have to see
  228| 	// the same number of frames the game actually rendered, and a test can drive
  229| 	// the transport with an arbitrary frame number. A process without a
  230| 	// SceneTree (the doctest binary runs the pump through this same function)
  231| 	// keeps the private counter so that the diagnostics still move.
  232| 	SceneTree *tree = SceneTree::get_singleton();
  233| 	if (tree != nullptr) {
  234| 		frame_count = (int)tree->get_frame();
  235| 	} else {
  236| 		frame_count++;
  237| 	}
  238| 	// TASK-044: the frame clock of an in-flight capture is handed in *before*
  239| 	// the transport poll, so a capture armed while a request is answered is
  240| 	// completed one frame later and not in the frame it was taken in.
  241| 	if (capture_engine != nullptr && capture_engine->is_active()) {
  242| 		capture_engine->begin_frame((int64_t)frame_count);
  243| 	}
  244| 	if (http_server != nullptr) {
  245| 		// Everything (socket polling, parsing, tool execution, advancing the
  246| 		// deferred tasks and writing the responses back) happens here, on the
  247| 		// main thread, once per frame.
  248| 		http_server->poll(max_requests_per_frame, (int64_t)frame_count);
  249| 	}
  250| 	// TASK-044: strictly after the responses of this frame. The encoding, the
  251| 	// PNGs, the pixel comparison and the appended log line of every capture
  252| 	// whose call was answered at least one rendered frame ago happen here - the
  253| 	// response was produced above and never waits for any of it.
  254| 	if (capture_engine != nullptr && capture_engine->is_active()) {
  255| 		capture_engine->tick((int64_t)frame_count);
  256| 	}
  257| }

  259| void MCPServer::_shutdown() {
  260| 	if (http_server != nullptr) {
  261| 		http_server->stop();
  262| 		memdelete(http_server);
  263| 		http_server = nullptr;
  264| 	}
  265| 	// TASK-038: the trace file is flushed and closed with everything else, so an
  266| 	// orderly shutdown never loses the last line. `close()` flushes first.
  267| 	if (trace_recorder != nullptr) {
  268| 		trace_recorder->close();
  269| 		memdelete(trace_recorder);
  270| 		trace_recorder = nullptr;
  271| 	}
  272| 	// TASK-044: the capture engine reports into the trace file, so it is torn
  273| 	// down next. In-flight captures are dropped rather than written: there is no
  274| 	// longer a file to describe them in.
  275| 	if (capture_engine != nullptr) {
  276| 		capture_engine->stop();
  277| 		memdelete(capture_engine);
  278| 		capture_engine = nullptr;
  279| 	}
  280| 	listening = false;
  281| 	port = 0;
  282| 	set_process(false);
  283| }

  285| void MCPServer::handle_jsonrpc_request(const String &p_body, MCPHttpOutcome &r_outcome) {
  286| 	// GDR-20: the dispatch either answers now or hands over a task for the
  287| 	// transport to advance; both are carried by the outcome.
  288| 	//
  289| 	// TASK-038: `p_trace` is true only while a recorder is open. With the trace
  290| 	// switched off (the default) the dispatcher is asked for exactly what it was
  291| 	// asked for before this task existed, and `r_outcome.trace` stays untraceable.
  292| 	const bool trace_enabled = trace_recorder != nullptr && trace_recorder->is_active();
  293| 
  294| 	// TASK-044: the whole cost this feature adds to a request is the one
  295| 	// framebuffer copy below, taken *before* the tool runs so that "before" means
  296| 	// "the frame the call was received in". `capture_token` is -1 for everything
  297| 	// that is not a captured `tools/call`, which is every request in the default
  298| 	// configuration.
  299| 	const bool capture_enabled = capture_engine != nullptr && capture_engine->is_active();
  300| 	int capture_token = -1;
  301| 	if (capture_enabled) {
  302| 		String capture_tool;
  303| 		if (MCPCapture::payload_calls_tool(p_body, capture_tool)) {
  304| 			if (registry.is_deferred_tool(StringName(capture_tool))) {
  305| 				// Answered across frames, so the call line is written from the
  306| 				// completion - far too late to decide whether the picture should
  307| 				// have been dropped. The log still gets a verdict instead of
  308| 				// silence; no snapshot is taken.
  309| 				capture_token = capture_engine->arm_unavailable(capture_tool,
  310| 						"the call is answered across frames (deferred), which the capture does not cover");
  311| 			} else {
  312| 				capture_token = capture_engine->arm(capture_tool);
  313| 			}
  314| 		}
  315| 	}
  316| 
  317| 	const MCPJsonRpc::Dispatch dispatched = MCPJsonRpc::dispatch(p_body, registry, is_editor, (uint64_t)pending_timeout_ms, trace_enabled);
  318| 	r_outcome.body = dispatched.response.body;
  319| 	r_outcome.http_status = dispatched.response.http_status;
  320| 	r_outcome.close = false;
  321| 	r_outcome.deferred = dispatched.deferred;
  322| 	r_outcome.task = dispatched.task;
  323| 	r_outcome.id_json = dispatched.id_json;
  324| 	r_outcome.timeout_ms = dispatched.timeout_ms;
  325| 	r_outcome.trace = dispatched.trace;
  326| 
  327| 	if (capture_enabled && capture_token >= 0) {
  328| 		// The call line is the *next* request line the recorder writes: the
  329| 		// transport writes it immediately after this call returns, with no other
  330| 		// line in between, so its `seq` is knowable here - and the capture line,
  331| 		// appended a frame later, carries the same number.
  332| 		const int capture_seq = trace_recorder->get_seq() + 1;
  333| 		if (capture_engine->finish(capture_token, dispatched.trace.ok, capture_seq)) {
  334| 			r_outcome.trace.capture_present = true;
  335| 			r_outcome.trace.capture_mode = MCPCapture::mode_name(capture_engine->get_config().mode);
  336| 			r_outcome.trace.capture_viewport = capture_engine->get_viewport_name();
  337| 			r_outcome.trace.capture_status = capture_engine->call_line_status(capture_token);
  338| 			r_outcome.trace.capture_reason = capture_engine->call_line_reason(capture_token);
  339| 			r_outcome.trace.capture_token = capture_token;
  340| 		}
  341| 	}
  342| }

  344| String MCPServer::build_deferred_body(const MCPDeferred::Completion &p_completion) {
  345| 	const String alias = p_name.replace("/", ".");
  346| 	if (alias != p_name) {
  347| 		names.push_back(alias);
  348| 	}
  349| 
  350| 	for (int i = 0; i < names.size(); i++) {
  351| 		if (!settings->has_setting(names[i])) {
  352| 			continue;
  353| 		}
  354| 		const Variant value = settings->get_setting(names[i]);
  355| 		if (value.get_type() == Variant::BOOL) {
  356| 			return (bool)value;
  357| 		}
  358| 	}
  359| 	return p_default;
  360| }

  362| void MCPServer::_register_tools() {
  363| 	if (registry.get_tool_count() > 0) {
  364| 		return;
  365| 	}

  367| 	// Every implemented group registers itself through the single shared entry
  368| 	// point; the group files own their tools (TASK-002 section 2.2.1). Tools that

  371| 	register_all_tools(registry);

  374| void MCPServer::_start_service() {
  375| 	if (service_started) {
  376| 		return;
  377| 	}
  378| 	service_started = true;
  379| 
  380| 	is_editor = Engine::get_singleton() != nullptr && Engine::get_singleton()->is_editor_hint();
  381| 
  382| 	_register_tools();
  383| 
  384| 	max_body_bytes = _get_int_setting("mcp_server/max_body_bytes", 8 * 1024 * 1024);
  385| 	max_requests_per_frame = _get_int_setting("mcp_server/max_requests_per_frame", 8);
  386| 	if (max_requests_per_frame < 1) {
  387| 		max_requests_per_frame = 1;
  388| 	}
  389| 	connection_idle_seconds = (double)_get_int_setting("mcp_server/connection_idle_seconds", 30);
  390| 	if (connection_idle_seconds < 1.0) {
  391| 		connection_idle_seconds = 1.0;
  392| 	}
  393| 
  394| 	// GDR-20: the ceiling of a deferred request (30 s by default) and the
  395| 	// per-frame tick budget. A budget below 1 would stall every pending request
  396| 	// forever, so it is clamped like `max_requests_per_frame`.
  397| 	//
  398| 	// TASK-014 R-3: a ceiling of 0 (or a negative value) is *not* a working
  399| 	// configuration. The transport reads 0 as "no deadline" and the frame-waiting
  400| 	// deferred tools declare no deadline of their own, so 0 would switch off the
  401| 	// module's only fallback for them. The M2 acceptance found the setting
  402| 	// accepted 0 silently (mcp_server.cpp used to fold a negative value into 0
  403| 	// and leave 0 alone); `MCPPendingTimeout::effective_ms` now maps `<= 0` to the
  404| 	// default and the startup line below reports both values.
  405| 	const int configured_pending_timeout_ms = _get_int_setting("mcp_server/pending_timeout_ms", MCPPendingTimeout::DEFAULT_MS);
  406| 	pending_timeout_ms = MCPPendingTimeout::effective_ms(configured_pending_timeout_ms);
  407| 	pending_ticks_per_frame = _get_int_setting("mcp_server/pending_ticks_per_frame", 8);
  408| 	if (pending_ticks_per_frame < 1) {
  409| 		pending_ticks_per_frame = 1;
  410| 	}
  411| 	if (pending_timeout_ms != configured_pending_timeout_ms) {
  412| 		WARN_PRINT(vformat("[MCP] mcp_server/pending_timeout_ms=%d is not a usable deadline (0 would switch the deferred fallback off); using %d ms",
  413| 				configured_pending_timeout_ms, pending_timeout_ms));
  414| 	}
  415| 	print_line(vformat("[MCP] pending_timeout_ms=%d (configured=%d) pending_ticks_per_frame=%d",
  416| 			pending_timeout_ms, configured_pending_timeout_ms, pending_ticks_per_frame));
  417| 
  418| 	const bool enabled_in_game = _get_bool_setting("godot_mcp/enabled_in_game", false);
  419| 
  420| 	ProjectSettings *settings = ProjectSettings::get_singleton();
  421| 	bool has_port_setting = false;
  422| 	int setting_port = 0;
  423| 	if (settings != nullptr && settings->has_setting("godot_mcp/port")) {
  424| 		const Variant value = settings->get_setting("godot_mcp/port");
  425| 		if (value.get_type() == Variant::INT || value.get_type() == Variant::FLOAT) {
  426| 			has_port_setting = true;
  427| 			setting_port = (int)value;
  428| 		}
  429| 	}
  430| 	// have not been ported yet are not registered at all, so they can never show
  431| 	// up in `tools/list` (GDR-7).
  432| 	register_all_tools(registry);
  433| }

  467| 	pending_timeout_ms = MCPPendingTimeout::effective_ms(configured_pending_timeout_ms);
  468| 	pending_ticks_per_frame = _get_int_setting("mcp_server/pending_ticks_per_frame", 8);
  469| 	if (pending_ticks_per_frame < 1) {
  470| 		pending_ticks_per_frame = 1;
  471| 	}
  472| 	if (pending_timeout_ms != configured_pending_timeout_ms) {
  473| 		WARN_PRINT(vformat("[MCP] mcp_server/pending_timeout_ms=%d is not a usable deadline (0 would switch the deferred fallback off); using %d ms",
  474| 				configured_pending_timeout_ms, pending_timeout_ms));
  475| 	}
  476| 	print_line(vformat("[MCP] pending_timeout_ms=%d (configured=%d) pending_ticks_per_frame=%d",
  477| 			pending_timeout_ms, configured_pending_timeout_ms, pending_ticks_per_frame));
  478| 
  479| 	const bool enabled_in_game = _get_bool_setting("godot_mcp/enabled_in_game", false);
  480| 
  481| 	ProjectSettings *settings = ProjectSettings::get_singleton();
  482| 	bool has_port_setting = false;
  483| 	int setting_port = 0;
  484| 	if (settings != nullptr && settings->has_setting("godot_mcp/port")) {
  485| 		const Variant value = settings->get_setting("godot_mcp/port");
  486| 		if (value.get_type() == Variant::INT || value.get_type() == Variant::FLOAT) {
  487| 			has_port_setting = true;
  488| 			setting_port = (int)value;
  489| 		}
  490| 	}
  491| 
  492| 	Vector<String> cmdline_args;
  493| 	if (OS::get_singleton() != nullptr) {
  494| 		const List<String> args = OS::get_singleton()->get_cmdline_args();
  495| 		for (const String &arg : args) {
  496| 			cmdline_args.push_back(arg);
  497| 		}
  498| 	}
  499| 
  500| 	// TASK-038: resolve the opt-in call trace exactly the way the port is
  501| 	// resolved (GDR-4): command line beats project setting beats the default, and
  502| 	// the default is "off" - no file is opened, and the transport is never given
  503| 	// a recorder, so nothing on the request path changes.
  504| 	{
  505| 		bool has_trace_setting = false;
  506| 		String setting_trace_path;
  507| 		if (settings != nullptr) {
  508| 			const char *const trace_setting_names[] = { "godot_mcp/trace_file", "godot_mcp.trace_file" };
  509| 			for (int i = 0; i < 2; i++) {
  510| 				if (!settings->has_setting(trace_setting_names[i])) {
  511| 					continue;
  512| 				}
  513| 				const Variant value = settings->get_setting(trace_setting_names[i]);
  514| 				if (value.get_type() == Variant::STRING) {
  515| 					has_trace_setting = true;
  516| 					setting_trace_path = (String)value;
  517| 					break;
  518| 				}
  519| 			}
  520| 		}
  521| 
  522| 		trace_config = MCPTrace::parse(cmdline_args, has_trace_setting, setting_trace_path);
  523| 
  524| 		if (trace_config.enabled) {
  525| 			trace_recorder = memnew(MCPTrace::Recorder);
  526| 			if (!trace_recorder->open(trace_config.path)) {
  527| 				// `open()` has already warned once; the recorder is dead but the
  528| 				// pointer stays valid (and inert) so `is_active()` is the single
  529| 				// question the transport asks.
  530| 				WARN_PRINT(vformat("[MCP] call trace could not be started for '%s'", trace_config.path));
  531| 			}
  532| 			if (trace_recorder->is_active()) {
  533| 				// The user has to be told that this file may contain project
  534| 				// content: it records the arguments of every tool call.
  535| 				print_line(vformat("[MCP] trace enabled: file=%s (records initialize / tools/list / every tools/call; arguments may contain project content)",
  536| 						trace_config.path));
  537| 			} else {
  538| 				print_line(vformat("[MCP] trace requested for '%s' but disabled (see the warning above)", trace_config.path));
  539| 			}
  540| 		} else {
  541| 			print_line("[MCP] trace=off (default; use --mcp-trace=<path> or godot_mcp/trace_file to enable)");
  542| 		}
  543| 	}
  544| 
  545| 	// TASK-044: the capture extension, resolved exactly like the trace above and
  546| 	// only ever *after* it - a capture line is a line in the trace file, so the
  547| 	// two switches are not independent, and a capture without a trace is refused
  548| 	// with a warning instead of being silently ignored.
  549| 	{
  550| 		bool has_capture_setting = false;
  551| 		String setting_capture_value;
  552| 		if (settings != nullptr) {
  553| 			const char *const capture_setting_names[] = { "godot_mcp/capture", "godot_mcp.capture" };
  554| 			for (int i = 0; i < 2; i++) {
  555| 				if (!settings->has_setting(capture_setting_names[i])) {
  556| 					continue;
  557| 				}
  558| 				const Variant value = settings->get_setting(capture_setting_names[i]);
  559| 				if (value.get_type() == Variant::STRING) {
  560| 					has_capture_setting = true;
  561| 					setting_capture_value = (String)value;
  562| 					break;
  563| 				}
  564| 			}
  565| 		}
  566| 
  567| 		capture_config = MCPCapture::parse(cmdline_args, has_capture_setting, setting_capture_value, trace_config.path);
  568| 		if (!capture_config.warning.is_empty()) {
  569| 			WARN_PRINT(vformat("[MCP] %s", capture_config.warning));
  570| 		}
  571| 
  572| 		if (capture_config.mode != MCPCapture::Mode::OFF) {
  573| 			MCPCapture::Engine *engine = memnew(MCPCapture::Engine);
  574| 			if (engine->start(capture_config, trace_recorder, is_editor)) {
  575| 				capture_engine = engine;
  576| 			} else {
  577| 				// `start()` has already warned; the engine holds nothing.
  578| 				memdelete(engine);
  579| 			}
  580| 		} else {
  581| 			print_line("[MCP] capture=off (default; use --mcp-capture=on_error|every_call together with --mcp-trace=<path>)");
  582| 		}
  583| 	}
  584| 
  585| 	port_config = MCPPort::parse(cmdline_args, has_port_setting, setting_port, is_editor);
  586| 	const bool listen = MCPPort::should_listen(is_editor, port_config, enabled_in_game);
  587| 
  588| 	const char *source = "default";
  589| 	if (port_config.explicit_cmdline) {
  590| 		source = "cmdline";
  591| 	} else if (port_config.from_project_setting) {
  592| 		source = "project_setting";
  593| 	}
  594| 
  595| 	print_line(vformat("[MCP] role=%s configured_port=%d source=%s listen=%s",
  596| 			is_editor ? "editor" : "game", port_config.port, source, listen ? "true" : "false"));
  597| 
  598| 	// TASK-054 (O-12): the generation marker of this trace file. The file is
  599| 	// append-only and may be reused across runs, so the observer needs one line
  600| 	// per process that says where a generation begins. It is written here
  601| 	// because this is the first place that knows all of its fields - the port is
  602| 	// resolved just above, the trace file was opened earlier - and it goes
  603| 	// through the event-line path, so it consumes no request `seq` (the first
  604| 	// request of this process is still `seq == 1`). It is written whether or not
  605| 	// the server ends up listening: the file was opened either way, and a run
  606| 	// that failed to bind is exactly the kind of fact a trace should carry.
  607| 	if (trace_recorder != nullptr) {
  608| 		trace_recorder->record_event_line(MCPTrace::build_trace_opened_fields(is_editor, port_config.port, listen));
  609| 	}
  610| 
  611| 	if (!listen) {
  612| 		port = 0;
  613| 		listening = false;
  614| 		print_line("[MCP] not listening (get_port()=0)");
  615| 		return;
  616| 	}
  617| 
  618| 	http_server = memnew(MCPHttpServer);
  619| 	http_server->set_sink(this);
  620| 	http_server->set_trace_recorder(trace_recorder);
  621| 	http_server->set_max_body_bytes(max_body_bytes);
  622| 	http_server->set_connection_idle_seconds(connection_idle_seconds);
  623| 	http_server->set_pending_timeout_ms((uint64_t)pending_timeout_ms);
  624| 	http_server->set_pending_ticks_per_frame(pending_ticks_per_frame);
  625| 
  626| 	const Error err = http_server->listen((uint16_t)port_config.port);
  627| 	if (err != OK) {
  628| 		// Binding failures must never take the engine down; the endpoint
  629| 		// simply stays disabled and get_port() reports 0.
  630| 		print_line(vformat("[MCP] bind failed on 127.0.0.1:%d (error=%d)", port_config.port, (int)err));
  631| 		WARN_PRINT(vformat("[MCP] bind failed on 127.0.0.1:%d; MCP server disabled", port_config.port));
  632| 		print_line("[MCP] get_port()=0 (MCP server disabled)");
  633| 
  634| 		memdelete(http_server);
  635| 		http_server = nullptr;
  636| 		port = 0;
  637| 		listening = false;
  638| 		return;
  639| 	}
  640| 
  641| 	port = port_config.port;
  642| 	listening = true;
  643| 	print_line(vformat("[MCP] listening on 127.0.0.1:%d (editor=%s, tools=%d)",
  644| 			port, is_editor ? "true" : "false", registry.get_visible_tool_count(is_editor)));
  645| 	// TASK-010 section 3.2: a *successful* bind gets an INFO line carrying the
  646| 	// port, the process role and where the port came from. The editor listens by
  647| 	// default (GDR-4), so an engine run that was never meant to serve MCP - the
  648| 	// obvious one being `--import`, which is an editor process and therefore
  649| 	// binds 9877 unless told otherwise - used to do so silently; the INFO line
  650| 	// makes that visible in the log next to the role line printed above.
  651| 	print_line(vformat("[MCP] INFO: MCP server is ready on 127.0.0.1:%d as the %s process (port source=%s, tools=%d)",
  652| 			port, is_editor ? "editor" : "game", source, registry.get_visible_tool_count(is_editor)));
  653| }
