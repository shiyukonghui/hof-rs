   44| 	if (p_suggestion.is_empty()) {

   52| MCPToolError MCPToolError::invalid_params(const String &p_message) {
   53| 	MCPToolError error;
   54| 	error.code = MCP_ERR_INVALID_PARAMS;
   55| 	// GDR-6: `-32602` carries the raw reason, never a prefix.
   56| 	//
   57| 	// TASK-050 O-1/N-7: the factory itself stays data-free - it does not know the
   58| 	// tool it is called for, and a handler must not have to repeat the schema it
   59| 	// was registered with. The one place that *does* know both is the registry
   60| 	// entry point, and it adds `data.suggestion` there (`call_tool()` /
   61| 	// `call_deferred_tool()`), so every `-32602` that leaves this module carries
   62| 	// one no matter which handler produced it.
   63| 	error.message = p_message;
   64| 	return error;
   65| }

   74| MCPToolError MCPToolError::no_scene() {
   75| 	MCPToolError error;
   76| 	error.code = MCP_ERR_TOOL_STATE;
   77| 	error.message = "No scene is currently open";
   78| 	// TASK-010 section 3.1: the advice has to match the *process*. "Use
   79| 	// editor_open_scene to open a scene first" is impossible advice in a game
   80| 	// process - there is no editor, and a game-only tool is not even served by an
   81| 	// editor endpoint - so a game-side caller is told what actually produces a
   82| 	// current scene there: the project's main scene, in a running game.
   83| 	//
   84| 	// The scope of the *tool* is deliberately not consulted: a game-only tool in
   85| 	// an editor process is refused with -32601 before any handler runs (GDR-7 /
   86| 	// GDR-19 section 17.3), so the editor branch is unreachable for it, and this
   87| 	// branch is unreachable in the game side of an editor *process*. What is left
   88| 	// is the honest question "which scene can this process have at all", and the
   89| 	// engine answers that: `Engine::is_editor_hint()`.
   90| 	Engine *engine = Engine::get_singleton();
   91| 	const bool editor_process = engine != nullptr && engine->is_editor_hint();
   92| 	error.data = _suggestion_data(editor_process
   93| 					? String("Use editor_open_scene to open a scene first")
   94| 					: String("Start the game with a main scene (application/run/main_scene in project.godot) and call this tool while the game is running"));
   95| 	return error;
   96| }

  119| MCPToolError MCPToolError::not_found(const String &p_what, const String &p_suggestion) {
  120| 	MCPToolError error;
  121| 	error.code = MCP_ERR_NOT_FOUND;
  122| 	// TASK-017 section 3: the " not found" suffix is appended **exactly once**.
  123| 	// A caller is allowed to spell the finding into `p_what` itself (the property
  124| 	// readback of `editor_get_node_properties` says "Property 'p' on node 'n' is
  125| 	// not readable by name"), and the unconditional suffix turned that into
  126| 	// "... is not readable by name not found" on the wire. Appending only when
  127| 	// the wording does not already end in the suffix keeps every other caller's
  128| 	// message byte-identical: their `p_what` never ends with the phrase, so the
  129| 	// concatenation is the same string it always was.
  130| 	error.message = p_what.ends_with(" not found") ? p_what : p_what + " not found";
  131| 	error.data = _suggestion_data(p_suggestion);
  132| 	return error;
  133| }

  153| // TASK-052 (DESIGN-DETAIL.md section 26 / GDR-28 point 4): the verbs an *added*
  154| // tool may introduce.
  155| //
  156| // GDR-28 requires an added entry to be named by the rule of GDR-16 and spells
  157| // out the channel restriction and the L4 `update_` ban, but it does not say

  159| // fixed the two names of TASK-052 section 1 (`project_build_csharp`,
  160| // `project_write_text_file`), so the two verbs have to be legal somewhere.
  161| //
  162| // The choice recorded here (and in REPORT-052) is the conservative one: the 37
  163| // word list above stays **byte for byte the map's set** - it is not edited - and
  164| // an added verb enters through this second, explicit list instead. So "which
  165| // verbs are the map's" and "which verbs were invented by an expansion" stay two
  166| // readable questions, and the same two records exist in
  167| // `scripts/gen_renamed_contract.py` (`ADDED_VERB_EXTENSIONS`) and in
  168| // `docs/tool-groups-added.json` (`verb_extensions`, where `--added` requires
  169| // each one to be used by a group).
  170| static const char *MCP_ADDED_TOOL_VERBS[] = { "build", "write" };

  172| static bool _is_closed_verb(const String &p_verb) {
  173| 	for (const char *verb : MCP_TOOL_VERBS) {
  174| 		if (p_verb == verb) {
  175| 			return true;
  176| 		}
  177| 	}
  178| 	for (const char *verb : MCP_ADDED_TOOL_VERBS) {
  179| 		if (p_verb == verb) {
  180| 			return true;
  181| 		}
  182| 	}
  183| 	return false;
  184| }

  186| bool MCPToolRegistry::parse_tool_name(const String &p_name, String &r_channel, String &r_verb) {
  187| 	for (const char *channel : MCP_TOOL_CHANNELS) {
  188| 		const String prefix = String(channel) + "_";
  189| 		if (!p_name.begins_with(prefix)) {
  190| 			continue;
  191| 		}
  192| 		const String rest = p_name.substr(prefix.length());
  193| 		if (rest.is_empty()) {
  194| 			return false;
  195| 		}
  196| 		// L1: the remainder may only contain [a-z0-9_].
  197| 		for (int i = 0; i < rest.length(); i++) {
  198| 			const char32_t c = rest[i];
  199| 			const bool allowed = (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '_';
  200| 			if (!allowed) {
  201| 				return false;
  202| 			}
  203| 		}
  204| 		const int separator = rest.find("_");
  205| 		r_channel = channel;
  206| 		// `editor_get` is legal: the object segment is not required (D-E4).
  207| 		r_verb = separator < 0 ? rest : rest.substr(0, separator);
  208| 		return false;
  209| 	}
  210| 	if (p_declared_channel != channel) {
  211| 		r_error = vformat("MCPToolRegistry: tool name '%s' declares channel '%s' but the name says '%s' (GDR-16 L3).", p_name, p_declared_channel, channel);
  212| 		return false;
  213| 	}
  214| 	if (p_declared_verb != verb) {
  215| 		r_error = vformat("MCPToolRegistry: tool name '%s' declares verb '%s' but the name says '%s' (GDR-16 L3).", p_name, p_declared_verb, verb);
  216| // scope / mutating to be declared and has applied the editor-process guard. The
  217| // lint below stays as the registry's own invariant, independent of its caller.
  218| bool MCPToolRegistry::register_tool(const MCPToolDef &p_def) {
  219| 	String reason;
  220| 	if (!validate_tool_name(String(p_def.name), p_def.channel, p_def.verb, reason)) {
  221| 		ERR_PRINT(reason);
  222| 		return false;
  223| 	}
  224| 	if (!tools.has(p_def.name)) {
  225| 		order.push_back(p_def.name);
  226| 	}
  227| 	tools[p_def.name] = p_def;
  228| 	return true;
  229| }

  284| Array MCPToolRegistry::build_tools_list(bool p_is_editor) const {
  285| 	Array list;
  286| 	for (int i = 0; i < order.size(); i++) {
  287| 		const MCPToolDef *def = tools.getptr(order[i]);
  288| 		if (def == nullptr || !scope_matches(def->scope, p_is_editor)) {
  289| 			// Tools that have not been ported yet are simply not in the
  290| 			// registry, and tools of the other process are hidden here.
  291| 			continue;
  292| 		}
  293| 		Dictionary entry;
  294| 		entry["name"] = String(def->name);
  295| 		entry["description"] = def->description;
  296| 		entry["inputSchema"] = def->input_schema;
  297| 		list.push_back(entry);
  298| 	}
  299| 	return list;
  300| }

  302| // ---------------------------------------------------------------------------
  303| // TASK-032 D4 (M4d): every argument a tool accepts is declared by its contract
  304| // `inputSchema`. A name outside that set is not "an extra the tool ignores" but
  305| // a caller's mistake: `project_get_settings` documents `prefix`, and a caller
  306| // who spelled it `filter` used to get `code: 0` plus the *unfiltered* list of 981
  307| // settings - a plausible-looking answer to a question that was never asked.
  308| //
  309| // The check lives here, in the registry, because this is the one entry point
  310| // every caller goes through: the immediate JSON-RPC path, the deferred channel
  311| // and the doctests all reach a handler through `call_tool()` /
  312| // `call_deferred_tool()`. A gate in the transport would leave the in-process
  313| // paths able to violate the rule.
  314| //
  315| // The message names the offending parameter(s) and `data.suggestion` lists the
  316| // accepted ones (in the schema's own order, so the answer is deterministic).
  317| // The check runs *before* the handler, which is also what makes the "accepted
  318| // set == schema" sweep in the TASK-032 doctest side-effect free: a call whose
  319| // only member is an undeclared name never reaches a handler.
  320| //
  321| // One boundary this gate does *not* cross (found by the TASK-032 doctest suite
  322| // itself): a parameter that the schema **does** declare but that the tool cannot
  323| // honour is the handler's business, not this one's.
  324| // `running_game_run_test_scenario.scene_path` is exactly that case - the
  325| // contract declares it (it is the editor-side migration source's argument), the
  326| // schema-driven gate therefore lets it through, and the handler still refuses a
  327| // non-empty value with its own explanation. The gate's rule is about *names no
  328| // schema declares*, and keeping those two rules apart is what lets a tool own a
  329| // semantic refusal without reopening the silent-ignore hole.
  330| // ---------------------------------------------------------------------------
  331| static bool _reject_unknown_arguments(const MCPToolDef &p_def, const Dictionary &p_args, MCPToolError &r_error) {
  332| 	const Variant properties_value = p_def.input_schema.get("properties", Variant());
  333| 	if (properties_value.get_type() != Variant::DICTIONARY) {
  334| 		// A tool whose schema declares no `properties` object cannot be checked
  335| 		// (the empty schema every no-parameter tool uses does declare one, so this
  336| 		// is a defect in a *schema literal*, not a silent bypass: it is reported
  337| 		// once, loudly, and the call proceeds as it did before TASK-032).
  338| 		if (!p_def.input_schema.is_empty()) {
  339| 			ERR_PRINT(vformat("MCPToolRegistry: tool '%s' has an inputSchema without a 'properties' object; "
  340| 							  "the unknown-argument gate cannot check its arguments",
  341| 					String(p_def.name)));
  342| 		}
  343| 		return true;
  344| 	}
  345| 	const Dictionary declared = properties_value;
  346| 
  347| 	Vector<String> unknown;
  348| 	const Array given = p_args.keys();
  349| 	for (int i = 0; i < given.size(); i++) {
  350| 		const String key = given[i];
  351| 		if (!declared.has(key)) {
  352| 			unknown.push_back(key);
  353| 		}
  354| 	}
  355| 	if (unknown.is_empty()) {
  356| 		return true;
  357| 	}
  358| 
  359| 	Vector<String> accepted;
  360| 	const Array accepted_keys = declared.keys();
  361| 	for (int i = 0; i < accepted_keys.size(); i++) {
  362| 		accepted.push_back((String)accepted_keys[i]);
  363| 	}
  364| 
  365| 	String message;
  366| 	if (unknown.size() == 1) {
  367| 		message = vformat("Unknown parameter '%s' for tool '%s'", unknown[0], String(p_def.name));
  368| 	} else {
  369| 		message = vformat("Unknown parameters %s for tool '%s'", String(", ").join(unknown), String(p_def.name));
  370| 
  371| // The accepted-parameter list of the TASK-032 gate and of every TASK-050
  372| // suggestion. A tool that declares nothing says so instead of listing an empty
  373| // set.
  374| static String _accepted_parameters_of(const MCPToolDef &p_def) {
  375| 	Dictionary declared;
  376| 	const Variant properties_value = p_def.input_schema.get("properties", Variant());
  377| 	if (properties_value.get_type() == Variant::DICTIONARY) {
  378| 		declared = properties_value;
  379| 	}
  380| 	Vector<String> accepted;
  381| 	const Array keys = declared.keys();
  382| 	for (int i = 0; i < keys.size(); i++) {
  383| 		accepted.push_back((String)keys[i]);
  384| 	}
  385| 	if (accepted.is_empty()) {
  386| 		return vformat("%s accepts no parameters", String(p_def.name));
  387| 	}
  388| 	return vformat("Accepted parameters of %s: %s", String(p_def.name), String(", ").join(accepted));
  389| }

  391| // A suggestion is a wire message, so every list it prints is bounded and says
  392| // when it was cut: an `Environment`-sized property bag must not be able to push
  393| // an unbounded `data` through the transport.
  394| static const int MCP_SUGGESTION_MEMBER_CAP = 12;
  395| 
  396| static String _schema_member_list(const Dictionary &p_properties) {
  397| 	const Array names = p_properties.keys();
  398| 	Vector<String> shown;
  399| 	for (int i = 0; i < names.size() && i < MCP_SUGGESTION_MEMBER_CAP; i++) {
  400| 		shown.push_back((String)names[i]);
  401| 	}
  402| 	String joined = String(", ").join(shown);
  403| 	if (names.size() > MCP_SUGGESTION_MEMBER_CAP) {
  404| 		joined += vformat(", ... (%d members in total)", names.size());
  405| 	}
  406| 	return joined;
  407| }

  409| static String _schema_required_list(const Variant &p_node) {
  410| 	if (p_node.get_type() != Variant::DICTIONARY) {
  411| 		return String();
  412| 	}
  413| 	const Variant required_value = ((Dictionary)p_node).get("required", Variant());
  414| 	if (required_value.get_type() != Variant::ARRAY) {
  415| 		return String();
  416| 	}
  417| 	const Array required = required_value;
  418| 	if (required.is_empty()) {
  419| 		return String();
  420| 	}
  421| 	Vector<String> names;
  422| 	for (int i = 0; i < required.size() && i < MCP_SUGGESTION_MEMBER_CAP; i++) {
  423| 		names.push_back((String)required[i]);
  424| 	}
  425| 	return String(", ").join(names);
  426| }

  428| // "a string" / "an integer" / "an array of an object with members {a, b}" ...
  429| static String _schema_type_phrase(const Variant &p_node, int p_depth) {
  430| 	if (p_node.get_type() != Variant::DICTIONARY) {
  431| 		return "any value";
  432| 	}
  433| 	const Dictionary node = p_node;
  434| 	const String type = node.get("type", String());
  435| 	if (type == "array") {
  436| 		const Variant items = node.get("items", Variant());
  437| 		if (items.get_type() == Variant::DICTIONARY && p_depth < 3) {
  438| 			return "an array of " + _schema_type_phrase(items, p_depth + 1);
  439| 		}
  440| }

  442| // The acceptable *form* of one declared parameter: its type and its `enum` when
  443| // the contract declares one. Requiredness is added by `_schema_form` below and
  444| // deliberately left out of the bare phrase: a "Missing required parameter"
  445| // refusal already says which parameter is missing, and repeating "(required)"
  446| // inside its own parentheses is noise.
  447| static String _schema_options_phrase(const Variant &p_node) {
  448| 	String phrase = _schema_type_phrase(p_node, 0);
  449| 	if (p_node.get_type() == Variant::DICTIONARY) {
  450| 		const Variant enum_value = ((Dictionary)p_node).get("enum", Variant());
  451| 		if (enum_value.get_type() == Variant::ARRAY) {
  452| 			const Array values = enum_value;
  453| 			Vector<String> spellings;
  454| 			for (int i = 0; i < values.size() && i < MCP_SUGGESTION_MEMBER_CAP; i++) {
  455| 				spellings.push_back(values[i].get_type() == Variant::STRING ? String(values[i]) : JSON::stringify(values[i]));
  456| 			}
  457| 			if (!spellings.is_empty()) {
  458| 				phrase += ", one of: " + String("|").join(spellings);
  459| 			}
  460| 		}
  461| 	}
  462| 	return phrase;
  463| }

  465| static String _schema_form(const Variant &p_node, bool p_required) {

  626| // `<name> must be ...` at the very start of a message, but only when the schema
  627| // really declares that name: the selection tool answers an unknown mode with
  628| // `mode must be one of: replace, add, remove` and names no parameter any other
  629| // way. Returning a declaration-checked name keeps this from inventing one.
  630| static String _declared_leading_word(const String &p_message, const Dictionary &p_properties) {
  631| 	const int space = p_message.find(" ");
  632| 	if (space <= 0) {
  633| 		return String();
  634| 	}
  635| 	const String candidate = p_message.substr(0, space);
  636| 	const Array names = p_properties.keys();
  637| 	for (int i = 0; i < names.size(); i++) {
  638| 		if (String(names[i]) == candidate) {
  639| 			return candidate;
  640| 		}
  641| 	}
  642| 	return String();

  645| static String _invalid_params_suggestion(const MCPToolDef &p_def, const String &p_message) {
  646| 	const String accepted = _accepted_parameters_of(p_def);
  647| 	Dictionary properties;
  648| 	const Variant properties_value = p_def.input_schema.get("properties", Variant());
  649| 	if (properties_value.get_type() == Variant::DICTIONARY) {
  650| 		properties = properties_value;
  651| 	}
  652| 	Array required;
  653| 	const Variant required_value = p_def.input_schema.get("required", Variant());
  654| 	if (required_value.get_type() == Variant::ARRAY) {
  655| 		required = required_value;
  656| 	}
  657| 
  658| 	String name = _parameter_name_in_message(p_message);
  659| 	if (name.is_empty()) {
  660| 		name = _declared_leading_word(p_message, properties);
  661| 	}
  662| 	if (name.is_empty()) {
  663| 		// N-7's last resort, and it is still not empty: say which tool's own rule
  664| 		// refused the call and list what it accepts, so the caller can move.
  665| 		return vformat("The refusal comes from %s's own value rule and names no member its inputSchema declares; %s",
  666| 				String(p_def.name), accepted);
  667| 	}
  668| 
  669| 	const bool missing = p_message.begins_with("Missing required parameter: ");
  670| 	const _SchemaPathLookup lookup = _lookup_schema_path(properties, required, name);
  671| 	if (lookup.resolved) {
  672| 		if (missing) {
  673| 			return vformat("Missing required parameter '%s' (%s); %s", name, _schema_options_phrase(lookup.node), accepted);
  674| 		}
  675| 		return vformat("Parameter '%s' accepts %s; %s", name, _schema_form(lookup.node, lookup.required), accepted);
  676| 	}
  677| 	if (!lookup.array_name.is_empty()) {
  678| 		return vformat("%s '%s': the schema declares '%s' as an array with no item shape, so no accepted member value can be "
  679| 					   "derived from it; %s",
  680| 				missing ? "Missing required parameter" : "Parameter", name, lookup.array_name, accepted);
  681| 	}
  682| 	if (missing) {
  683| 		return vformat("Missing required parameter '%s'; %s", name, accepted);
  684| 	}
  685| 	return vformat("Parameter '%s' is not declared by %s's inputSchema; the refusal is the tool's own value rule; %s",
  686| 			name, String(p_def.name), accepted);
  687| }

  689| // Adds the suggestion to a `-32602` that does not have one. Everything else -
  690| // code, message, and any `data` a handler already attached - is preserved.
  691| static void _add_invalid_params_suggestion(const MCPToolDef &p_def, MCPToolError &r_error) {
  692| 	if (r_error.code != MCP_ERR_INVALID_PARAMS || _has_non_empty_suggestion(r_error.data)) {
  693| 		return;
  694| 	}
  695| 	Dictionary data;
  696| 	if (r_error.data.get_type() == Variant::DICTIONARY) {
  697| 		data = r_error.data;
  698| 	}
  699| 	data["suggestion"] = _invalid_params_suggestion(p_def, r_error.message);
  700| 	r_error.data = data;

  710| // ---------------------------------------------------------------------------
  711| // TASK-032 D4 (M4d): every argument a tool accepts is declared by its contract
  712| // `inputSchema`. A name outside that set is not "an extra the tool ignores" but
  713| // a caller's mistake: `project_get_settings` documents `prefix`, and a caller
  714| // who spelled it `filter` used to get `code: 0` plus the *unfiltered* list of 981
  715| // settings - a plausible-looking answer to a question that was never asked.
  716| //
  717| // The check lives here, in the registry, because this is the one entry point
  718| // every caller goes through: the immediate JSON-RPC path, the deferred channel
  719| // and the doctests all reach a handler through `call_tool()` /
  720| // `call_deferred_tool()`. A gate in the transport would leave the in-process
  721| // paths able to violate the rule.
  722| //
  723| // The message names the offending parameter(s) and `data.suggestion` lists the
  724| // accepted ones (in the schema's own order, so the answer is deterministic).
  725| // The check runs *before* the handler, which is also what makes the "accepted
  726| // set == schema" sweep in the TASK-032 doctest side-effect free: a call whose
  727| // only member is an undeclared name never reaches a handler.
  728| //
  729| // One boundary this gate does *not* cross (found by the TASK-032 doctest suite
  730| // itself): a parameter that the schema **does** declare but that the tool cannot
  731| // honour is the handler's business, not this one's.
  732| // `running_game_run_test_scenario.scene_path` is exactly that case - the
  733| // contract declares it (it is the editor-side migration source's argument), the
  734| // schema-driven gate therefore lets it through, and the handler still refuses a
  735| // non-empty value with its own explanation. The gate's rule is about *names no
  736| // schema declares*, and keeping those two rules apart is what lets a tool own a
  737| // semantic refusal without reopening the silent-ignore hole.
  738| // ---------------------------------------------------------------------------
  739| static bool _reject_unknown_arguments(const MCPToolDef &p_def, const Dictionary &p_args, MCPToolError &r_error) {
  740| 	const Variant properties_value = p_def.input_schema.get("properties", Variant());
  741| 	if (properties_value.get_type() != Variant::DICTIONARY) {
  742| 		// A tool whose schema declares no `properties` object cannot be checked
  743| 		// (the empty schema every no-parameter tool uses does declare one, so this
  744| 		// is a defect in a *schema literal*, not a silent bypass: it is reported
  745| 		// once, loudly, and the call proceeds as it did before TASK-032).
  746| 		if (!p_def.input_schema.is_empty()) {
  747| 			ERR_PRINT(vformat("MCPToolRegistry: tool '%s' has an inputSchema without a 'properties' object; "
  748| 							  "the unknown-argument gate cannot check its arguments",
  749| 					String(p_def.name)));
  750| 		}
  751| 		return true;
  752| 	}
  753| 	const Dictionary declared = properties_value;
  754| 
  755| 	Vector<String> unknown;
  756| 	const Array given = p_args.keys();
  757| 	for (int i = 0; i < given.size(); i++) {
  758| 		const String key = given[i];
  759| 		if (!declared.has(key)) {
  760| 			unknown.push_back(key);
  761| 		}
  762| 	}
  763| 	if (unknown.is_empty()) {
  764| 		return true;
  765| 	}
  766| 
  767| 	String message;
  768| 	if (unknown.size() == 1) {
  769| 		message = vformat("Unknown parameter '%s' for tool '%s'", unknown[0], String(p_def.name));
  770| 	} else {
  771| 		message = vformat("Unknown parameters %s for tool '%s'", String(", ").join(unknown), String(p_def.name));
  772| 	}
  773| 	// The one list both this gate and the TASK-050 suggestions print, so the two
  774| 	// can never drift apart.
  775| 	const String suggestion = _accepted_parameters_of(p_def);
  776| 
  777| 	r_error = MCPToolError::invalid_params(message);
  778| 	r_error.data = _suggestion_data(suggestion);
  779| 	return false;
  780| }

  782| Variant MCPToolRegistry::call_tool(const StringName &p_name, const Dictionary &p_args, MCPToolError &r_error) const {
  783| 	const MCPToolDef *def = tools.getptr(p_name);
  784| 	if (def == nullptr) {
  785| 		r_error = MCPToolError::invalid_params(vformat("Unknown tool: %s", String(p_name)));
  786| 		r_error.data = _unknown_tool_suggestion_data(p_name);
  787| 		return Variant();
  788| 	}
  789| 	if (!_reject_unknown_arguments(*def, p_args, r_error)) {
  790| 		return Variant();
  791| 	}
  792| 	if (def->is_deferred()) {
  793| 		// GDR-20: a deferred tool cannot be answered by a single call. Refusing
  794| 		// here is what keeps the one-frame path honest - silently ticking once
  795| 		// would answer a lie (N samples of the same frame), which is exactly the
  796| 		// defect TASK-011 exists to remove. The transport routes these calls
  797| 		// through `call_deferred_tool()`.
  798| 		r_error = MCPToolError::internal(vformat("Tool '%s' answers across frames and must be called through the deferred channel", String(p_name)));
  799| 		return Variant();
